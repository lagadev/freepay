Authentication Bearer Key

প্রতিটা Brand-এর নিজস্ব API key আছে (Brand পেজে পাবেন)। ইনভয়েস তৈরি ও SMS ingest — দুটোতেই এই key লাগবে:

Authorization: Bearer YOUR_BRAND_API_KEY

Payment Initiation Node.js

const express = require('express');
const axios = require('axios');
const db = require('./db');
const app = express();
app.use(express.urlencoded({ extended: true }));

app.post('/pay', async (req, res) => {
    const { amount } = req.body;
    const order_id = "ORD-" + Date.now();

    const api_key = "YOUR_BRAND_API_KEY";
    const api_url = "https://your-worker.workers.dev/api/invoices";

    try {
        const response = await axios.post(api_url, {
            amount,
            reference: order_id,
            callbackUrl: "https://yourdomain.com/callback",
            redirectUrl: "https://yourdomain.com/success?id=" + order_id,
        }, {
            headers: { Authorization: `Bearer ${api_key}` }
        });

        await db.execute("INSERT INTO payments (order_id, amount) VALUES (?, ?)", [order_id, amount]);
        res.redirect(response.data.payUrl);
    } catch (error) {
        res.status(500).send("API Error");
    }
});

Webhook Listener Node.js

ইনভয়েস verified হলে callbackUrl-এ signed POST যায়। ব্যালেন্স/অর্ডার আপডেট করার আগে অবশ্যই X-Signature ভেরিফাই করুন।

const crypto = require('crypto');
app.use(express.json());

app.post('/callback', async (req, res) => {
    const raw = JSON.stringify(req.body);
    const expected = crypto.createHmac('sha256', process.env.BRAND_API_KEY)
        .update(raw).digest('hex');

    if (req.headers['x-signature'] !== expected) {
        return res.status(401).send('bad signature');
    }

    const { event, order_id, status, trxId } = { event: req.body.event, order_id: req.body.reference, status: 'paid', trxId: req.body.trxId };
    if (event === 'invoice.verified') {
        const [rows] = await db.execute("SELECT status FROM payments WHERE order_id = ?", [order_id]);
        if (rows.length && rows[0].status !== 'paid') {
            await db.execute("UPDATE payments SET status='paid', txn_id=? WHERE order_id=?", [trxId, order_id]);
        }
    }
    res.sendStatus(200);
});

Invoice Status GET

GET /api/invoices/:id

পাবলিক endpoint। status: pending / verified / expired।

Select Payment Method POST

POST /api/invoices/:id/select-method
{ "method": "bkash" }

Verify Transaction ID POST

POST /api/invoices/:id/verify
{ "trxId": "8N7A1B2C3D", "senderNumber": "01712345678" }

TrxID শুধু তখনই ম্যাচ করবে যদি এটি এই ইনভয়েসের নিজের ১৫ মিনিটের উইন্ডোর মধ্যে আসা SMS-এর সাথে মেলে — অর্থাৎ শেষ ১৫ মিনিটে "verify" হওয়া লাগবে।
