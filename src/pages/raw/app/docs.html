Authentication Bearer Key

প্রতিটা Brand-এর নিজস্ব Invoice API Key আছে (Brand পেজে পাবেন)। ইনভয়েস তৈরি করতে এই key লাগবে:

Authorization: Bearer YOUR_INVOICE_API_KEY

Integration আপনার প্ল্যাটফর্ম বেছে নিন

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;
use App\Models\Payment;

class PaymentController extends Controller
{
    public function pay(Request $request)
    {
        $order_id = "ORD-" . time();
        $amount = $request->amount;

        $response = Http::withToken(env('FREEPAY_API_KEY'))
            ->post(env('FREEPAY_API_URL'), [
                'amount'      => $amount,
                'reference'   => $order_id,
                'callbackUrl' => route('freepay.callback'),
                'redirectUrl' => route('freepay.success', ['id' => $order_id]),
            ]);

        $res = $response->json();

        if ($response->successful() && !empty($res['payUrl'])) {
            Payment::create(['order_id' => $order_id, 'amount' => $amount]);
            return redirect($res['payUrl']);
        }
        return back()->with('error', $res['error'] ?? 'Payment failed');
    }
}
04 WEBHOOK (routes/api.php)
Route::post('/callback', function (Request $request) {
    $raw = $request->getContent();
    $expected = hash_hmac('sha256', $raw, env('FREEPAY_API_KEY'));

    if (!hash_equals($expected, $request->header('X-Signature', ''))) {
        return response('bad signature', 401);
    }

    $data = $request->json()->all();
    if (($data['event'] ?? '') === 'invoice.verified') {
        $payment = \App\Models\Payment::where('order_id', $data['reference'])->first();
        if ($payment && $payment->status !== 'paid') {
            $payment->update(['status' => 'paid', 'txn_id' => $data['trxId']]);
        }
    }
    return response('OK', 200);
})->name('freepay.callback');

const express = require('express');
const axios = require('axios');

app.post('/pay', async (req, res) => {
    const { amount } = req.body;
    const order_id = "ORD-" + Date.now();

        if (response.data.payUrl) {
            await db.execute("INSERT INTO payments (order_id, amount) VALUES (?, ?)", [order_id, amount]);
            res.redirect(response.data.payUrl);
        } else {
            res.send("Error: " + response.data.error);
        }
    } catch (error) {
        res.status(500).send("API Error");
    }
});
04 WEBHOOK LISTENER
const crypto = require('crypto');
app.use(express.json());

app.post('/callback', async (req, res) => {
    const raw = JSON.stringify(req.body);
    const expected = crypto.createHmac('sha256', API_KEY).update(raw).digest('hex');

    if (req.headers['x-signature'] !== expected) {
        return res.status(401).send('bad signature');
    }

    const { event, reference, trxId } = req.body;
    if (event === 'invoice.verified') {
        const [rows] = await db.execute("SELECT status FROM payments WHERE order_id = ?", [reference]);
        if (rows.length && rows[0].status !== 'paid') {
            await db.execute("UPDATE payments SET status='paid', txn_id=? WHERE order_id=?", [trxId, reference]);
        }
    }
    res.sendStatus(200);
});

03 INITIATE PAYMENT
function initiate_freepay_payment($amount) {
    global $wpdb;
    $order_id = "ORD-" . time();
    $table_name = $wpdb->prefix . 'freepay_payments';

    $response = wp_remote_post(FREEPAY_API_URL, [
        'headers' => [
            'Authorization' => 'Bearer ' . FREEPAY_API_KEY,
            'Content-Type'  => 'application/json',
        ],
        'body' => json_encode([
            'amount'      => $amount,
            'reference'   => $order_id,
            'callbackUrl' => home_url('/wp-json/freepay/v1/callback'),
            'redirectUrl' => home_url('/payment-success?id=' . $order_id),
        ]),
    ]);

    if (!is_wp_error($response)) {
        $body = json_decode(wp_remote_retrieve_body($response), true);
        if (!empty($body['payUrl'])) {
            $wpdb->insert($table_name, ['order_id' => $order_id, 'amount' => $amount]);
            wp_redirect($body['payUrl']); exit;
        }
    }
}
04 WEBHOOK LISTENER (REST API)
add_action('rest_api_init', function () {
    register_rest_route('freepay/v1', '/callback', [
        'methods'  => 'POST',
        'callback' => 'freepay_handle_callback',
        'permission_callback' => '__return_true',
    ]);
});

function freepay_handle_callback($request) {
    global $wpdb;
    $raw = $request->get_body();
    $expected = hash_hmac('sha256', $raw, FREEPAY_API_KEY);

    if (!hash_equals($expected, $request->get_header('x_signature') ?? '')) {
        return new WP_REST_Response('bad signature', 401);
    }

    $data = json_decode($raw, true);
    $table_name = $wpdb->prefix . 'freepay_payments';

    if ($data && $data['event'] === 'invoice.verified') {
        $order_id = $data['reference'];
        $txn_id   = $data['trxId'];
        $order = $wpdb->get_row($wpdb->prepare("SELECT status FROM $table_name WHERE order_id = %s", $order_id));

        if ($order && $order->status !== 'paid') {
            $wpdb->update($table_name, ['status' => 'paid', 'txn_id' => $txn_id], ['order_id' => $order_id]);
        }
        return new WP_REST_Response('SUCCESS', 200);
    }
    return new WP_REST_Response('IGNORED', 200);
}

@app.route('/pay', methods=['POST'])
def pay():
    amount = request.form.get('amount')
    order_id = f"ORD-{int(time.time())}"

    payload = {
        "amount": amount,
        "reference": order_id,
        "callbackUrl": "https://yourdomain.com/callback",
        "redirectUrl": f"https://yourdomain.com/success?id={order_id}",
    }
    headers = {"Authorization": f"Bearer {API_KEY}"}

    try:
        response = requests.post(API_URL, json=payload, headers=headers)
        res_data = response.json()

        if res_data.get("payUrl"):
            sql = "INSERT INTO payments (order_id, amount) VALUES (%s, %s)"
            cursor.execute(sql, (order_id, amount))
            db.commit()
            return redirect(res_data["payUrl"])
        return f"Error: {res_data.get('error')}"
    except Exception as e:
        return str(e)

@app.route('/callback', methods=['POST'])
def callback():
    raw = request.get_data()
    expected = hmac.new(API_KEY.encode(), raw, hashlib.sha256).hexdigest()

    if request.headers.get('X-Signature') != expected:
        return "bad signature", 401

    data = request.json
    if data and data.get('event') == 'invoice.verified':
        order_id = data.get('reference')
        txn_id = data.get('trxId')

        cursor.execute("SELECT status FROM payments WHERE order_id = %s", (order_id,))
        order = cursor.fetchone()

        if order and order['status'] != 'paid':
            cursor.execute("UPDATE payments SET status='paid', txn_id=%s WHERE order_id=%s", (txn_id, order_id))
            db.commit()
        return "SUCCESS", 200

    return "IGNORED", 200

if __name__ == '__main__':
    app.run(debug=True)

TrxID শুধু তখনই ম্যাচ করবে যদি এটি এই ইনভয়েসের নিজের ১৫ মিনিটের উইন্ডোর মধ্যে আসা SMS-এর সাথে মেলে — অর্থাৎ শেষ ১৫ মিনিটে "verify" হওয়া লাগবে।
